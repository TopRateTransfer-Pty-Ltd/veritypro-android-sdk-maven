STEP 1 — Create VpBrandConfig:

iOS (CREATE: verity-pro-ios/VerityPro/models/VpBrandConfig.swift):
```swift
import SwiftUI
public struct VpBrandConfig: Sendable {
    public let primaryColor: String?
    public let logoUrl: String?
    public init(primaryColor: String? = nil, logoUrl: String? = nil) {
        self.primaryColor = primaryColor
        self.logoUrl = logoUrl
    }
    func resolvedPrimaryColor() -> Color? {
        guard let hex = primaryColor, !hex.isEmpty else { return nil }
        // parse "#RRGGBB" or "#RGB" → Color
        var str = hex.trimmingCharacters(in: .whitespaces)
        if str.hasPrefix("#") { str = String(str.dropFirst()) }
        guard str.count == 6 || str.count == 3 else {
            print("[VpBrandConfig] WARNING: invalid primaryColor '\(hex)', using default")
            return nil
        }
        let full = str.count == 3 ? str.flatMap { [$0, $0] }.map(String.init).joined() : str
        guard let value = UInt64(full, radix: 16) else {
            print("[VpBrandConfig] WARNING: could not parse hex '\(hex)', using default")
            return nil
        }
        return Color(red: Double((value >> 16) & 0xFF)/255,
                     green: Double((value >> 8) & 0xFF)/255,
                     blue: Double(value & 0xFF)/255)
    }
    func resolvedLogoUrl() -> URL? {
        guard let raw = logoUrl, !raw.isEmpty else { return nil }
        guard let url = URL(string: raw), url.scheme == "https" else {
            print("[VpBrandConfig] WARNING: logoUrl must be HTTPS, logo will not be shown")
            return nil
        }
        return url
    }
}
```

Android (ADD to verity_options.kt):
```kotlin
@Parcelize
data class VpBrandConfig(
    val primaryColor: String? = null,
    val logoUrl: String? = null
) : Parcelable {
    fun resolvedPrimaryColor(): androidx.compose.ui.graphics.Color? {
        val hex = primaryColor?.trim()?.trimStart('#') ?: return null
        return try {
            val value = java.lang.Long.parseLong(
                if (hex.length == 3) hex.flatMap { listOf(it, it) }.joinToString("") else hex,
                16
            )
            androidx.compose.ui.graphics.Color(
                red = ((value shr 16) and 0xFF).toFloat() / 255f,
                green = ((value shr 8) and 0xFF).toFloat() / 255f,
                blue = (value and 0xFF).toFloat() / 255f
            )
        } catch (e: NumberFormatException) {
            android.util.Log.w("VpBrandConfig", "Invalid primaryColor '$primaryColor', using default")
            null
        }
    }
    fun resolvedLogoUrl(): android.net.Uri? {
        val raw = logoUrl ?: return null
        val uri = android.net.Uri.parse(raw)
        if (uri.scheme != "https") {
            android.util.Log.w("VpBrandConfig", "logoUrl must be HTTPS, logo will not be shown")
            return null
        }
        return uri
    }
}
```
